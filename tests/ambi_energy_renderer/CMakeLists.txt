cmake_minimum_required(VERSION 3.20)
project(s3g_ambi_energy_renderer_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
include(CTest)
get_filename_component(repo "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(renderer "${repo}/plugins/clap_ambisonic_energy_visualizer/renderer")
add_executable(s3g_ambi_energy_renderer_probe probe.cpp "${renderer}/energy_reference.cpp")
target_compile_features(s3g_ambi_energy_renderer_probe PRIVATE cxx_std_17)
target_include_directories(s3g_ambi_energy_renderer_probe PRIVATE "${renderer}" "${repo}/dsp")
if(APPLE)
  enable_language(OBJCXX)
  target_sources(s3g_ambi_energy_renderer_probe PRIVATE "${renderer}/energy_metal.mm")
  set_source_files_properties("${renderer}/energy_metal.mm" PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
  target_link_libraries(s3g_ambi_energy_renderer_probe PRIVATE "-framework Foundation" "-framework Metal")
elseif(WIN32)
  target_sources(s3g_ambi_energy_renderer_probe PRIVATE "${renderer}/energy_d3d.cpp")
  target_link_libraries(s3g_ambi_energy_renderer_probe PRIVATE d3d11 dxgi d3dcompiler)
  if(MINGW)
    target_link_options(s3g_ambi_energy_renderer_probe PRIVATE -municode)
  endif()
else()
  message(FATAL_ERROR "This GPU prototype currently supports Metal/macOS and D3D11/Windows")
endif()
if(MSVC)
  target_compile_options(s3g_ambi_energy_renderer_probe PRIVATE /W4 /permissive-)
else()
  target_compile_options(s3g_ambi_energy_renderer_probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME energy_reference_unit COMMAND s3g_ambi_energy_renderer_probe --self-test)
